    t = rng.uniform(0.1, 2 * np.pi - 0.1)        # t lejos de 0 (mod 2 pi)
    w = rng.normal(size=2)
    A = np.eye(2) - rot(t)
    assert np.isclose(np.linalg.det(A), 2 - 2 * np.cos(t), atol=TOL)
    c = np.linalg.solve(A, w)                    # único, porque det(A) > 0
